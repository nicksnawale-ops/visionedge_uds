// Stacked bar chart — Breakfast / Lunch / Snacks / Dinner. Only the top segment is rounded.
// Full version: src/components/charts/StackedBarChart.tsx
const days = ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'];
const series = [
  { label: 'Breakfast', color: '#0B2D6B', values: [620, 604, 642, 628, 612, 412, 386] },
  { label: 'Lunch', color: '#1E63FF', values: [840, 832, 868, 856, 838, 604, 552] },
  { label: 'Snacks', color: '#00C6FF', values: [310, 302, 328, 318, 306, 226, 204] },
  { label: 'Dinner', color: '#8C95A6', values: [540, 546, 564, 554, 542, 400, 376] },
];

const W = 600, H = 260;
const M = { top: 12, right: 16, bottom: 28, left: 48 };
const MAX = 2500;
const band = (W - M.left - M.right) / days.length;
const barW = Math.min(band * 0.56, 48);
const y = (v: number) => H - M.bottom - (v / MAX) * (H - M.top - M.bottom);
const topRounded = (x: number, top: number, w: number, h: number, r = 4) =>
  `M${x},${top + h}V${top + r}A${r},${r} 0 0 1 ${x + r},${top}H${x + w - r}A${r},${r} 0 0 1 ${x + w},${top + r}V${top + h}Z`;

export function MealsByTypeStackedBar() {
  return (
    <div className="bg-white p-6 rounded-[12px] border border-[#DDE2EA]">
      <div className="flex flex-wrap items-start justify-between gap-x-6 gap-y-2 mb-4">
        <div>
          <h3 className="text-[16px] font-semibold text-[#121722]">Meals by type per day</h3>
          <p className="text-[12px] text-[#4B5465]">Meal Pulse · 24–30 Sep 2026</p>
        </div>
        <ul className="flex flex-wrap gap-x-4 gap-y-1">
          {series.map((s) => (
            <li key={s.label} className="flex items-center gap-1.5 text-[12px] text-[#353D4C]">
              <span className="w-2 h-2 rounded-full" style={{ backgroundColor: s.color }} />
              {s.label}
            </li>
          ))}
        </ul>
      </div>

      <svg viewBox={`0 0 ${W} ${H}`} className="w-full" role="img" aria-label="Lunch is the largest meal every day; Wed had the highest total: 2,402.">
        {[0, 500, 1000, 1500, 2000, 2500].map((t) => (
          <line key={t} x1={M.left} x2={W - M.right} y1={y(t)} y2={y(t)} stroke="#DDE2EA" strokeWidth={1} />
        ))}
        {days.map((day, i) => {
          const x = M.left + band * i + band / 2 - barW / 2;
          let base = 0;
          return series.map((s, si) => {
            const top = y(base + s.values[i]);
            const bottom = y(base);
            base += s.values[i];
            const d = si === series.length - 1
              ? topRounded(x, top, barW, bottom - top)
              : `M${x},${top}h${barW}V${bottom}h${-barW}Z`;
            return <path key={`${day}-${s.label}`} d={d} fill={s.color} tabIndex={0} aria-label={`${day}, ${s.label}: ${s.values[i]} meals`} />;
          });
        })}
      </svg>
    </div>
  );
}
